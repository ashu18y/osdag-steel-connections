\section{Introduction}
This report presents the manual verification of a beam-to-column end plate connection designed using Osdag. The verification is carried out as per IS 800:2007 provisions.

\section{Input Data}
\begin{itemize}
    \item Beam Section: MB 250
    \item Column Section: HB 450
    \item Material: Fe 410 ($f_y = 250$ MPa, $f_u = 410$ MPa)
    \item Applied Moment: 5 kNm
    \item Applied Shear: 5 kN
    \item Bolt: 8 mm diameter, Grade 4.6 ($f_{ub} = 400$ MPa)
\end{itemize}

\section{Load Adjustment (IS 800 Cl. 10.7)}
Minimum load requirements:

\begin{align}
M_u &= 0.5 M_d = 0.5 \times 105.68 = 52.84 \text{ kNm} \\
V_u &= 0.15 V_d = 18.33 \text{ kN}
\end{align}

\textbf{Adopted Loads:}
\begin{itemize}
    \item Moment = 52.84 kNm
    \item Shear = 18.33 kN
\end{itemize}

\section{Bolt Design}

\subsection{Assumptions}
\begin{itemize}
    \item Bolts in single shear
    \item Threads in shear plane
    \item $\gamma_{mb} = 1.25$
\end{itemize}

\subsection{Bolt Shear Capacity}
Net area $A_{nb} = 36.6 \, mm^2$

\begin{equation}
V_{dsb} = \frac{f_{ub} \cdot A_{nb}}{\sqrt{3} \cdot \gamma_{mb}}
= \frac{400 \times 36.6}{\sqrt{3} \times 1.25 \times 1000}
= 6.76 \, \text{kN}
\end{equation}

\subsection{Bolt Bearing Capacity}

\begin{equation}
k_b = \min\left(\frac{e}{3d_0}, \frac{p}{3d_0} - 0.25, \frac{f_u}{f_{ub}}, 1.0\right)
\end{equation}

\begin{equation}
k_b = \min\left(\frac{15}{24}, \frac{30}{24}-0.25, \frac{410}{400}, 1.0\right) = 0.62
\end{equation}

\begin{equation}
V_{dpb} = \frac{2.5 \cdot k_b \cdot d \cdot t \cdot f_u}{\gamma_{mb}}
= \frac{2.5 \times 0.62 \times 8 \times 14 \times 410}{1.25 \times 1000}
= 56.94 \, \text{kN}
\end{equation}

\subsection{Governing Capacity}

\begin{equation}
V_{db} = \min(6.76, 56.94) = 6.76 \, \text{kN}
\end{equation}

\section{Bolt Force Check}

\subsection{Shear per Bolt}

\begin{equation}
V_{sb} = \frac{V_u}{n} = \frac{18.33}{24} = 0.76 \, \text{kN}
\end{equation}

\subsection{Tension in Bolt}

\begin{equation}
T_b = 9.65 \, \text{kN}
\end{equation}

\subsection{Interaction Check}

\begin{equation}
\left(\frac{V_{sb}}{V_{db}}\right)^2 + \left(\frac{T_b}{T_{db}}\right)^2
= \left(\frac{0.76}{6.76}\right)^2 + \left(\frac{9.65}{10.54}\right)^2
= 0.85 < 1
\end{equation}

\textbf{Safe}

\section{Prying Force}

\begin{equation}
Q = 0
\end{equation}

Plate thickness is sufficient to prevent prying action.

\section{End Plate Design}

\subsection{Moment at Critical Section}

\begin{equation}
M_{cr} = T_b \cdot l_v = 9.65 \times 8.5 \times 10^{-3} = 0.08 \, \text{kNm}
\end{equation}

\subsection{Required Thickness}

\begin{equation}
t_p = \sqrt{\frac{4M_{cr}}{b_e (f_y/\gamma_{m0})}} = 6.8 \, \text{mm}
\end{equation}

\textbf{Provided thickness = 14 mm (Safe)}

\section{Weld Design}

\subsection{Required Weld Size}

\begin{equation}
t_w = \frac{V_u}{f_u \cdot L_w} \times \sqrt{3} \gamma_{mw} = 0.39 \, \text{mm}
\end{equation}

\textbf{Provided weld = 6 mm (Safe)}

\section{Observations}
\begin{itemize}
    \item Manual calculations match Osdag results accurately.
    \item Shear governs bolt design.
    \item Load adjustment significantly increases design forces.
\end{itemize}

\section{Bugs and Recommendations}
\begin{itemize}
    \item Osdag automatically modifies loads as per IS 800 Clause 10.7 without clear visibility.
    \item Plate thickness optimization is not transparent.
    \item Recommendation: Provide step-by-step calculation details in output.
\end{itemize}

\section{Conclusion}
Manual verification confirms that the Osdag design is accurate and complies with IS 800:2007. The connection is safe under the given loading conditions.
